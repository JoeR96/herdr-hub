# herdr-hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A cross-platform herdr plugin that keeps chosen project spaces open, writes branch / agent / PR / release status onto sidebar space rows, creates a worktree with Claude on a hotkey, and offers a utilities menu including workstream cleanup.

**Architecture:** A herdr plugin folder (`herdr-plugin.toml`) whose commands are Node 24 TypeScript files run directly (no build). Pure decision logic lives in `src/model/` and is unit-tested with fixtures; thin wrappers in `src/io/` call `herdr` (via `HERDR_BIN_PATH`), `git` and `gh` with `execFile`. A detached background daemon refreshes sidebar tokens; popups use a small raw-mode TTY list widget.

**Tech Stack:** Node 24 (built-in type stripping, `node:test`), TypeScript (erasable syntax only), herdr 0.9 plugin API, git, GitHub CLI.

**Spec:** `docs/superpowers/specs/2026-10-10-herdr-hub-design.md`

## Global Constraints

- Node >= 24. Source files are `.ts` run directly by Node: erasable syntax only (no `enum`, `namespace`, parameter properties, decorators). Type-only imports MUST use `import type`. Relative imports MUST include the `.ts` extension.
- No npm dependencies at all (runtime or dev). `package.json` has no `dependencies`/`devDependencies`.
- External programs are started only with `execFile`/`spawn` and argument arrays, never through a shell: herdr via `HERDR_BIN_PATH` (fallback `herdr`), `git`, `gh`.
- Must work on Windows, macOS and Linux. Never hard-code `\` or `/` path separators for filesystem work; use `node:path`.
- Plugin id `herdr-hub`. Metadata source id `herdr-hub`. Sidebar tokens are `hub1`..`hub4`.
- Indent character is U+2800 (BRAILLE PATTERN BLANK); herdr trims ordinary leading whitespace (verified 2026-10-10).
- Never `git worktree remove --force`; never `git branch -D`.
- Tests run with `npm test` = `node --test "test/**/*.test.ts"`.
- Test fixtures are synthetic. Never commit real data captured from private repos.

## Review Focus

- Paths for the same folder that differ only by a `\\?\` prefix, slash direction or letter case (herdr vs git on Windows) must still match: pinned in Task 1 (`normPath`) and Task 4 (`classify` with herdr-style paths).
- Popup keys arriving as `ESC O A` / `ESC O B` (application cursor mode) or `\r\n` for enter must still work: pinned in Task 9 (`parseKeys`).
- A burst of `workspace.closed` events must not create duplicate pinned spaces: pinned in Task 7 (`withFileLock` serialises) and Task 8 (`missingPinned`).
- A repo with no `origin`, a non-GitHub `origin`, or no `gh` must show local lines only and never crash: pinned in Task 2 (`parseGitHubRemote`), Task 4 (`buildLines` with `ghProblem`) and Task 7 (`ghStatus` missing).
- A freshly created worktree whose tip equals the base tip must not be offered for cleanup as "merged": pinned in Task 6.

```ts
import { execFile } from "node:child_process";

export interface RunResult { code: number; stdout: string; stderr: string; missing: boolean }

// Runs a program with an argument array (never a shell). Never throws: failures come back in the result.
export function run(cmd: string, args: string[], opts: { cwd?: string; timeoutMs?: number } = {}): Promise<RunResult> {
  return new Promise((resolve) => {
    execFile(cmd, args, {
      cwd: opts.cwd,
      timeout: opts.timeoutMs ?? 30_000,
      maxBuffer: 32 * 1024 * 1024,
      windowsHide: true,
      encoding: "utf8",
    }, (err, stdout, stderr) => {
      const e = err as ({ code?: number | string; message?: string } | null);
      const missing = e?.code === "ENOENT";
      const code = e ? (typeof e.code === "number" ? e.code : 1) : 0;
      resolve({ code, stdout: stdout ?? "", stderr: stderr || (e?.message ?? ""), missing });
    });
  });
}
```

`src/model/paths.ts`:

```ts
// herdr reports Windows paths with a \\?\ prefix; git reports C:/... with forward slashes.
export function cleanPath(p: string): string {
  return p.replace(/^\\\\\?\\/, "");
}

export function normPath(p: string, platform: string = process.platform): string {
  let s = cleanPath(p).replace(/\\/g, "/");
  while (s.length > 1 && s.endsWith("/") && !/^[A-Za-z]:\/$/.test(s)) s = s.slice(0, -1);
  return platform === "win32" ? s.toLowerCase() : s;
}

export function samePath(a: string, b: string, platform: string = process.platform): boolean {
  return normPath(a, platform) === normPath(b, platform);
}
```

```ts
export function age(fromMs: number, nowMs: number): string {
  const s = Math.max(0, Math.round((nowMs - fromMs) / 1000));
  if (s < 60) return "just now";
  const m = Math.floor(s / 60);
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const d = Math.floor(h / 24);
  if (d < 14) return `${d}d ago`;
  return `${Math.floor(d / 7)}w ago`;
}
```

`src/model/git-parse.ts`:
```ts
export interface GitStatus { branch: string | null; upstream: string | null; ahead: number; behind: number; dirty: boolean }
export interface WorktreeEntry { path: string; head: string; branch: string | null; prunable: boolean; isMain: boolean }
export interface GitHubRepo { owner: string; name: string }

export function parseStatusV2(text: string): GitStatus {
  const st: GitStatus = { branch: null, upstream: null, ahead: 0, behind: 0, dirty: false };
  for (const line of text.split(/\r?\n/)) {
    if (!line) continue;
    if (line.startsWith("# branch.head ")) {
      const h = line.slice("# branch.head ".length);
      st.branch = h === "(detached)" ? null : h;
    } else if (line.startsWith("# branch.upstream ")) {
      st.upstream = line.slice("# branch.upstream ".length);
    } else if (line.startsWith("# branch.ab ")) {
      const m = /\+(\d+) -(\d+)/.exec(line);
      if (m) { st.ahead = Number(m[1]); st.behind = Number(m[2]); }
    } else if (!line.startsWith("#")) {
      st.dirty = true;
    }
  }
  return st;
}

export function parseWorktreeList(text: string): WorktreeEntry[] {
  const out: WorktreeEntry[] = [];
  text.split(/\r?\n\r?\n/).forEach((block, index) => {
    const lines = block.split(/\r?\n/).filter(Boolean);
    const wt = lines.find((l) => l.startsWith("worktree "));
    if (!wt || lines.includes("bare")) return;
    const branch = lines.find((l) => l.startsWith("branch "))?.slice("branch ".length) ?? null;
    out.push({
      path: wt.slice("worktree ".length),
      head: lines.find((l) => l.startsWith("HEAD "))?.slice("HEAD ".length) ?? "",
      branch: branch ? branch.replace(/^refs\/heads\//, "") : null,
      prunable: lines.some((l) => l === "prunable" || l.startsWith("prunable ")),
      isMain: index === 0,
    });
  });
  return out;
}

export function parseGitHubRemote(url: string): GitHubRepo | null {
  const m = /^(?:https?:\/\/(?:[^@/]+@)?github\.com\/|git@github\.com:|ssh:\/\/git@github\.com\/)([^/\s]+)\/([^/\s]+?)(?:\.git)?\/?$/.exec(url.trim());
  return m ? { owner: m[1], name: m[2] } : null;
}

export function parseRevParse(text: string): { top: string; commonDir: string } | null {
  const [top, commonDir] = text.split(/\r?\n/).filter(Boolean);
  return top && commonDir ? { top, commonDir } : null;
}

export function repoRootFromCommonDir(commonDir: string): string {
  return commonDir.replace(/[\\/]\.git[\\/]?$/, "");
}
```

`src/io/git.ts`:
```ts
import { statSync } from "node:fs";
import { join } from "node:path";
import { run } from "./exec.ts";
import type { RunResult } from "./exec.ts";
import { parseStatusV2, parseWorktreeList, parseRevParse, parseGitHubRemote, repoRootFromCommonDir } from "../model/git-parse.ts";
import type { GitStatus, WorktreeEntry, GitHubRepo } from "../model/git-parse.ts";

export interface GitLocation { top: string; repoRoot: string }

function git(cwd: string, args: string[]): Promise<RunResult> {
  return run("git", ["-C", cwd, ...args], { timeoutMs: 60_000 });
}

export async function locate(dir: string): Promise<GitLocation | null> {
  const r = await git(dir, ["rev-parse", "--path-format=absolute", "--show-toplevel", "--git-common-dir"]);
  if (r.code !== 0) return null;
  const p = parseRevParse(r.stdout);
  return p ? { top: p.top, repoRoot: repoRootFromCommonDir(p.commonDir) } : null;
}

export async function status(dir: string): Promise<GitStatus | null> {
  const r = await git(dir, ["status", "--porcelain=v2", "--branch"]);
  return r.code === 0 ? parseStatusV2(r.stdout) : null;
}

export async function githubRepo(repo: string): Promise<GitHubRepo | null> {
  const r = await git(repo, ["remote", "get-url", "origin"]);
  return r.code === 0 ? parseGitHubRemote(r.stdout) : null;
}

export async function worktrees(repo: string): Promise<WorktreeEntry[]> {
  const r = await git(repo, ["worktree", "list", "--porcelain"]);
  return r.code === 0 ? parseWorktreeList(r.stdout) : [];
}

export async function defaultBranch(repo: string): Promise<string> {
  const r = await git(repo, ["symbolic-ref", "--short", "refs/remotes/origin/HEAD"]);
  const ref = r.stdout.trim();
  return r.code === 0 && ref ? ref.replace(/^origin\//, "") : "main";
}

// origin/<default> when it exists locally, otherwise HEAD
export async function baseRef(repo: string): Promise<string> {
  const b = await defaultBranch(repo);
  const r = await git(repo, ["rev-parse", "--verify", "--quiet", `refs/remotes/origin/${b}`]);
  return r.code === 0 ? `origin/${b}` : "HEAD";
}

export async function tipOf(repo: string, ref: string): Promise<string | null> {
  const r = await git(repo, ["rev-parse", "--verify", "--quiet", ref]);
  return r.code === 0 ? r.stdout.trim() : null;
}

export async function mergedBranches(repo: string, base: string): Promise<Set<string>> {
  const r = await git(repo, ["branch", "--merged", base, "--format=%(refname:short)"]);
  return new Set(r.code === 0 ? r.stdout.split(/\r?\n/).map((s) => s.trim()).filter(Boolean) : []);
}

export async function headTime(dir: string): Promise<number | null> {
  const r = await git(dir, ["log", "-1", "--format=%ct"]);
  const n = Number(r.stdout.trim());
  return r.code === 0 && n > 0 ? n * 1000 : null;
}

// A linked worktree's .git file is written when the worktree is created.
export function createdTime(worktreePath: string): number | null {
  try { return statSync(join(worktreePath, ".git")).mtimeMs; } catch { return null; }
}

export async function fetchOrigin(repo: string): Promise<void> {
  await git(repo, ["fetch", "--quiet", "origin"]);
}

export function removeWorktree(repo: string, path: string): Promise<RunResult> {
  return git(repo, ["worktree", "remove", path]);
}

export function pruneWorktrees(repo: string): Promise<RunResult> {
  return git(repo, ["worktree", "prune"]);
}

export function deleteBranch(repo: string, branch: string): Promise<RunResult> {
  return git(repo, ["branch", "-d", branch]);
}
```

```ts
export type CheckState = "success" | "failure" | "pending" | "none";

// GitHub's statusCheckRollup.state already combines every check on the commit.
export function rollupState(s?: string | null): CheckState {
  switch (s) {
    case "SUCCESS": return "success";
    case "FAILURE": case "ERROR": return "failure";
    case "PENDING": case "EXPECTED": return "pending";
    default: return "none";
  }
}

export function runState(status: string, conclusion: string | null): CheckState {
  if (status !== "completed") return "pending";
  if (conclusion === "success") return "success";
  if (["failure", "timed_out", "startup_failure", "action_required", "cancelled"].includes(conclusion ?? "")) return "failure";
  return "none";
}

export function deploymentState(s?: string | null): CheckState {
  switch (s) {
    case "SUCCESS": case "INACTIVE": return "success";
    case "FAILURE": case "ERROR": return "failure";
    case "IN_PROGRESS": case "QUEUED": case "PENDING": case "WAITING": return "pending";
    default: return "none";
  }
}

export function symbol(s: CheckState): string {
  return { success: "✓", failure: "✗", pending: "⋯", none: "" }[s];
}
```

```ts
import { deploymentState, runState } from "./checks.ts";
import type { CheckState } from "./checks.ts";

export interface Release { source: "deployment" | "release" | "workflow"; name: string; state: CheckState; atMs: number; url: string | null }
export interface DeploymentNode { environment: string; createdAt: string; latestStatus: { state: string; environmentUrl?: string | null; logUrl?: string | null } | null }
export interface ReleaseNode { name: string | null; tagName: string; url: string; publishedAt: string | null; isDraft: boolean }
export interface RunNode { name: string; path: string; status: string; conclusion: string | null; html_url: string; created_at: string }

function latest<T>(items: T[], at: (t: T) => number): T | undefined {
  return items.slice().sort((a, b) => at(b) - at(a))[0];
}

export function pickRelease(deployments: DeploymentNode[], releases: ReleaseNode[], runs: RunNode[], pattern: RegExp): Release | null {
  const candidates: Release[] = [];

  const production = deployments.filter((d) => /^production$/i.test(d.environment));
  const dep = latest(production.length ? production : deployments, (d) => Date.parse(d.createdAt));
  if (dep) {
    candidates.push({
      source: "deployment", name: dep.environment, state: deploymentState(dep.latestStatus?.state),
      atMs: Date.parse(dep.createdAt), url: dep.latestStatus?.environmentUrl ?? dep.latestStatus?.logUrl ?? null,
    });
  }

  const rel = latest(releases.filter((r) => !r.isDraft && r.publishedAt), (r) => Date.parse(r.publishedAt!));
  if (rel) {
    candidates.push({ source: "release", name: rel.name || rel.tagName, state: "success", atMs: Date.parse(rel.publishedAt!), url: rel.url });
  }

  const run = latest(runs.filter((r) => pattern.test(r.name) || pattern.test(r.path)), (r) => Date.parse(r.created_at));
  if (run) {
    candidates.push({ source: "workflow", name: run.name, state: runState(run.status, run.conclusion), atMs: Date.parse(run.created_at), url: run.html_url });
  }

  return latest(candidates, (c) => c.atMs) ?? null;
}
```

```ts
import { rollupState } from "./checks.ts";
import type { CheckState } from "./checks.ts";
import { pickRelease } from "./release.ts";
import type { Release } from "./release.ts";

export interface PullRequest { number: number; title: string; url: string; isDraft: boolean; headRefName: string; checks: CheckState }
export interface RepoData { defaultBranch: string | null; prs: PullRequest[]; release: Release | null }

export function mapRepo(gql: any, runs: any, pattern: RegExp): RepoData {
  const repo = gql?.data?.repository;
  if (!repo) throw new Error(gql?.errors?.[0]?.message ?? "repository not found");
  const prs: PullRequest[] = (repo.pullRequests?.nodes ?? []).map((n: any) => ({
    number: n.number,
    title: n.title,
    url: n.url,
    isDraft: !!n.isDraft,
    headRefName: n.headRefName,
    checks: rollupState(n.commits?.nodes?.[0]?.commit?.statusCheckRollup?.state),
  }));
  const release = pickRelease(repo.deployments?.nodes ?? [], repo.releases?.nodes ?? [], runs?.workflow_runs ?? [], pattern);
  return { defaultBranch: repo.defaultBranchRef?.name ?? null, prs, release };
}
```

```ts
import { cleanPath, normPath, samePath } from "./paths.ts";

export interface WorkspaceLike { label: string; worktree?: { repo_root?: string; checkout_path?: string; is_linked_worktree?: boolean } | null }
export interface Location { top: string; repoRoot: string }
export type SpaceKind =
  | { kind: "main" | "worktree"; repoRoot: string; repoKey: string; folder: string }
  | { kind: "other" };

// herdr only records repo info after a worktree command touched the space, so git locations are the fallback.
export function classify(ws: WorkspaceLike, locations: (Location | null)[]): SpaceKind {
  const wt = ws.worktree;
  if (wt?.repo_root) {
    const repoRoot = cleanPath(wt.repo_root);
    if (wt.is_linked_worktree && wt.checkout_path) {
      return { kind: "worktree", repoRoot, repoKey: normPath(repoRoot), folder: cleanPath(wt.checkout_path) };
    }
    if (!wt.is_linked_worktree) return { kind: "main", repoRoot, repoKey: normPath(repoRoot), folder: repoRoot };
  }
  const loc = locations.find((l): l is Location => l !== null);
  if (!loc) return { kind: "other" };
  return {
    kind: samePath(loc.top, loc.repoRoot) ? "main" : "worktree",
    repoRoot: loc.repoRoot,
    repoKey: normPath(loc.repoRoot),
    folder: loc.top,
  };
}
```

```ts
import type { GitStatus } from "./git-parse.ts";
import type { PullRequest } from "./github-map.ts";
import type { Release } from "./release.ts";
import { symbol } from "./checks.ts";
import { age } from "./format.ts";

// herdr trims ordinary leading whitespace from token values; U+2800 survives and renders blank.
export const BLANK = "\u2800";
export const TOKEN_NAMES = ["hub1", "hub2", "hub3", "hub4"] as const;

export type IndentStyle = "blank" | "bar";
export interface AgentInfo { kind: string; title: string; state: string }
export interface SpaceView {
  kind: "main" | "worktree";
  git: GitStatus | null;
  agents: AgentInfo[];
  pr: PullRequest | null;
  release: Release | null;
  ghProblem: string | null;
  stale: boolean;
  now: number;
  indent: IndentStyle;
}

function indentFor(kind: SpaceView["kind"], style: IndentStyle): string {
  return style === "bar" ? "│ " : BLANK.repeat(kind === "main" ? 4 : 6);
}

function branchLine(g: GitStatus | null): string | null {
  if (!g) return null;
  let s = `${g.branch ?? "detached"} ${g.dirty ? "●" : "✓"}`;
  if (g.ahead) s += ` ⇡${g.ahead}`;
  if (g.behind) s += ` ⇣${g.behind}`;
  return s;
}

function releaseLine(r: Release, now: number, stale: boolean): string {
  const sym = symbol(r.state);
  return [`⇡ ${r.name}`, ...(sym ? [sym] : []), age(r.atMs, now), ...(stale ? ["stale"] : [])].join(" · ");
}

function prLine(p: PullRequest, stale: boolean): string {
  return [`PR #${p.number}`, p.isDraft ? "draft" : "open", `checks ${symbol(p.checks) || "none"}`, ...(stale ? ["stale"] : [])].join(" · ");
}

function agentLine(agents: AgentInfo[]): string | null {
  if (!agents.length) return null;
  const a = agents[0];
  return `${a.kind} · ${a.title} · ${a.state}${agents.length > 1 ? ` +${agents.length - 1}` : ""}`;
}

export function buildLines(v: SpaceView): string[] {
  const raw = v.kind === "main"
    ? [branchLine(v.git), v.release ? releaseLine(v.release, v.now, v.stale) : null, v.ghProblem]
    : [agentLine(v.agents), branchLine(v.git), v.pr ? prLine(v.pr, v.stale) : null];
  const pad = indentFor(v.kind, v.indent);
  return raw.filter((l): l is string => !!l).slice(0, TOKEN_NAMES.length).map((l) => pad + l);
}

export function toTokens(lines: string[]): Record<string, string | null> {
  const tokens: Record<string, string | null> = {};
  TOKEN_NAMES.forEach((name, i) => { tokens[name] = lines[i] ?? null; });
  return tokens;
}
```

```ts
export function defaults(): Config {
  return {
    pinned: [],
    refresh: { localSeconds: 15, githubSeconds: 180 },

    releaseWorkflowPattern: "release|deploy|publish",
    newWorktree: { branchPrefix: "feature/" },

export function parseConfig(text: string | null): { config: Config; errors: string[] } {
  const config = defaults();
  const errors: string[] = [];
  if (text === null || !text.trim()) return { config, errors };
  let raw: any;
  try { raw = JSON.parse(text); } catch (e) {
    return { config, errors: [`config.json is not valid JSON: ${(e as Error).message}`] };
  }

  if (Array.isArray(raw?.pinned)) {
    for (const p of raw.pinned) {
      const ok = p && typeof p.label === "string" && p.label.trim() && typeof p.path === "string" && p.path.trim();
      if (!ok) { errors.push(`pinned entry ignored: ${JSON.stringify(p)}`); continue; }
      if (config.pinned.some((q) => q.label === p.label.trim())) { errors.push(`duplicate pinned label ignored: ${p.label}`); continue; }
      config.pinned.push({ label: p.label.trim(), path: p.path.trim() });
    }
  } else if (raw?.pinned !== undefined) {
    errors.push("pinned must be a list");
  }

  const num = (v: unknown, name: string, min: number, fallback: number): number => {
    if (v === undefined) return fallback;
    if (typeof v === "number" && Number.isFinite(v) && v >= min) return v;
    errors.push(`${name} must be a number >= ${min}`);
    return fallback;
  };
  config.refresh.localSeconds = num(raw?.refresh?.localSeconds, "refresh.localSeconds", 5, 15);
  config.refresh.githubSeconds = num(raw?.refresh?.githubSeconds, "refresh.githubSeconds", 30, 180);
  config.cleanup.staleDays = num(raw?.cleanup?.staleDays, "cleanup.staleDays", 1, 30);

  if (raw?.releaseWorkflowPattern !== undefined) {
    try { new RegExp(String(raw.releaseWorkflowPattern), "i"); config.releaseWorkflowPattern = String(raw.releaseWorkflowPattern); }
    catch { errors.push("releaseWorkflowPattern is not a valid regular expression"); }
  }
  if (typeof raw?.newWorktree?.branchPrefix === "string") config.newWorktree.branchPrefix = raw.newWorktree.branchPrefix;
  if (raw?.indent === "blank" || raw?.indent === "bar") config.indent = raw.indent;
  else if (raw?.indent !== undefined) errors.push('indent must be "blank" or "bar"');

  return { config, errors };
}

export function configDir(): string {
  return process.env.HERDR_PLUGIN_CONFIG_DIR ?? join(homedir(), ".herdr-hub");
}

export function configFile(dir: string): string {
  return join(dir, "config.json");
}

export function loadConfig(dir: string): { config: Config; errors: string[] } {
  const file = configFile(dir);
  return parseConfig(existsSync(file) ? readFileSync(file, "utf8") : null);
}

export function saveConfig(dir: string, cfg: Config): void {
  mkdirSync(dir, { recursive: true });
  writeFileSync(configFile(dir), JSON.stringify(cfg, null, 2) + "\n");
}

export function togglePin(cfg: Config, label: string, path: string): { config: Config; pinned: boolean } {
  const has = cfg.pinned.some((p) => p.label === label);
  const pinned = has ? cfg.pinned.filter((p) => p.label !== label) : [...cfg.pinned, { label, path }];
  return { config: { ...cfg, pinned }, pinned: !has };
}
```

```ts
export function slugify(s: string): string {
  return s.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "");
}

export function initials(label: string): string {
  const s = label.split(/[^A-Za-z0-9]+/).filter(Boolean).map((w) => w[0]).join("").toLowerCase();
  return s || "wt";
}

// "stairs fix" -> feature/stairs-fix; "fix/stairs" keeps the typed branch.
export function planWorktree(typed: string, branchPrefix: string, spaceLabel: string): { branch: string; label: string; agentName: string } | null {
  const label = typed.trim();
  if (!label) return null;
  const hasSlash = label.includes("/");
  const slug = slugify(hasSlash ? label.slice(label.lastIndexOf("/") + 1) : label);
  if (!slug) return null;
  const branch = hasSlash
    ? label.slice(0, label.lastIndexOf("/") + 1).toLowerCase().replace(/\s+/g, "-") + slug
    : branchPrefix + slug;
  let agentName = `${initials(spaceLabel)}-${slug}`;
  if (!/^[a-z]/.test(agentName)) agentName = `c-${agentName}`;
  agentName = agentName.slice(0, 32).replace(/-+$/, "");
  return { branch, label, agentName };
}
```

```ts
import type { WorktreeEntry } from "./git-parse.ts";
import { normPath } from "./paths.ts";

export type Reason = "missing" | "merged" | "pr-merged" | "stale";
export interface CleanupInput {
  worktrees: WorktreeEntry[];
  baseTip: string | null;
  merged: Set<string>;
  mergedPrs: Map<string, number>;
  openPrBranches: Set<string>;
  activityMs: Map<string, number>;
  dirty: Map<string, boolean>;
  spaces: Map<string, { workspaceId: string; agentCount: number }>;
  staleDays: number;
  now: number;
}
export interface Candidate {
  path: string;
  branch: string | null;
  reason: Reason;
  detail: string;
  dirty: boolean;
  workspaceId: string | null;
  agentCount: number;
  selectable: boolean;
}

const ORDER: Reason[] = ["missing", "merged", "pr-merged", "stale"];
const DAY = 86_400_000;

export function findCandidates(i: CleanupInput): Candidate[] {
  const out: Candidate[] = [];
  for (const w of i.worktrees) {
    if (w.isMain) continue;
    const key = normPath(w.path);
    const space = i.spaces.get(key);
    const common = { path: w.path, branch: w.branch, workspaceId: space?.workspaceId ?? null, agentCount: space?.agentCount ?? 0 };

    if (w.prunable) {
      out.push({ ...common, reason: "missing", detail: "folder missing", dirty: false, selectable: true });
      continue;
    }

    const hasOpenPr = !!w.branch && i.openPrBranches.has(w.branch);
    let reason: Reason | null = null;
    let detail = "";
    // a worktree still sitting on the base tip has no work of its own yet: it's new, not merged
    if (w.branch && i.merged.has(w.branch) && w.head !== i.baseTip) {
      reason = "merged"; detail = "merged";
    } else if (w.branch && i.mergedPrs.has(w.branch) && !hasOpenPr) {
      reason = "pr-merged"; detail = `PR #${i.mergedPrs.get(w.branch)} merged`;
    } else if (!hasOpenPr) {
      const last = i.activityMs.get(key);
      const days = last === undefined ? 0 : Math.floor((i.now - last) / DAY);
      if (days >= i.staleDays) { reason = "stale"; detail = `stale ${days} days`; }
    }
    if (!reason) continue;

    const dirty = i.dirty.get(key) ?? false;
    out.push({ ...common, reason, detail, dirty, selectable: !dirty });
  }
  return out.sort((a, b) => ORDER.indexOf(a.reason) - ORDER.indexOf(b.reason) || a.path.localeCompare(b.path));
}

export function warnings(c: Candidate): string[] {
  const w: string[] = [];
  if (c.dirty) w.push("dirty: has uncommitted changes");
  if (c.agentCount) w.push("agent running: will be stopped");
  return w;
}
```

```ts
const MAX_LOG = 1024 * 1024;

export function log(p: StatePaths, msg: string): void {
  try {
    if (existsSync(p.log) && statSync(p.log).size > MAX_LOG) renameSync(p.log, p.log + ".1");
    appendFileSync(p.log, `${new Date().toISOString()} ${msg}\n`);
  } catch { /* logging must never break the plugin */ }
}

export function lastLogLines(p: StatePaths, n: number): string[] {
  try { return readFileSync(p.log, "utf8").trimEnd().split("\n").slice(-n); } catch { return []; }
}

export function pidAlive(pid: number): boolean {
  try { process.kill(pid, 0); return true; } catch (e) { return (e as { code?: string }).code === "EPERM"; }
}

export function readPidLock(file: string): number | null {
  try {
    const n = Number(readFileSync(file, "utf8").trim());
    return Number.isInteger(n) && n > 0 ? n : null;
  } catch { return null; }
}

export function acquirePidLock(file: string): boolean {
  const pid = readPidLock(file);
  if (pid && pid !== process.pid && pidAlive(pid)) return false;
  writeFileSync(file, String(process.pid));
  return true;
}

export function releasePidLock(file: string): void {
  if (readPidLock(file) === process.pid) { try { unlinkSync(file); } catch { /* already gone */ } }
}

// Cross-process mutex: whoever creates the file first owns it. Locks older than staleMs are broken.
export async function withFileLock<T>(file: string, fn: () => Promise<T>, timeoutMs = 30_000, staleMs = 60_000): Promise<T> {
  const start = Date.now();
  for (;;) {
    try { closeSync(openSync(file, "wx")); break; } catch (e) {
      if ((e as { code?: string }).code !== "EEXIST") throw e;
      try {
        if (Date.now() - statSync(file).mtimeMs > staleMs) { unlinkSync(file); continue; }
      } catch { continue; }
      if (Date.now() - start > timeoutMs) throw new Error(`timed out waiting for ${file}`);
      await new Promise((r) => setTimeout(r, 150));
    }
  }
  try { return await fn(); } finally { try { unlinkSync(file); } catch { /* already gone */ } }
}

export function touchTrigger(p: StatePaths): void {
  writeFileSync(p.trigger, String(Date.now()));
}

export interface CacheEntry { data: RepoData | null; fetchedAt: number; error: string | null }
export interface Cache { repos: Record<string, CacheEntry> }

export function readCache(p: StatePaths): Cache {
  try {
    const c = JSON.parse(readFileSync(p.cache, "utf8"));
    return c && typeof c.repos === "object" && c.repos !== null ? c : { repos: {} };
  } catch { return { repos: {} }; }
}

export function writeCache(p: StatePaths, c: Cache): void {
  const tmp = p.cache + ".tmp";
  writeFileSync(tmp, JSON.stringify(c));
  try { renameSync(tmp, p.cache); } catch { writeFileSync(p.cache, JSON.stringify(c)); }
}
```

```ts
export class HerdrError extends Error {}

export class Herdr {
  bin: string;
  prefix: string[];

  constructor(bin?: string, prefix: string[] = []) {
    this.bin = bin ?? process.env.HERDR_BIN_PATH ?? "herdr";
    this.prefix = prefix;
  }

  async call(args: string[]): Promise<any> {
    const r = await run(this.bin, [...this.prefix, ...args], { timeoutMs: 20_000 });
    const text = r.stdout.trim();
    let json: any = null;
    try { json = text ? JSON.parse(text) : null; } catch { /* not JSON */ }
    if (json?.error) throw new HerdrError(json.error.message ?? JSON.stringify(json.error));
    if (r.code !== 0) throw new HerdrError(`herdr ${args.slice(0, 2).join(" ")} failed: ${(r.stderr || r.stdout).trim()}`);
    return json?.result ?? null;
  }

  async snapshot(): Promise<Snapshot> {
    const s = (await this.call(["api", "snapshot"]))?.snapshot ?? {};
    return { ...s, workspaces: s.workspaces ?? [], panes: s.panes ?? [], agents: s.agents ?? [] };
  }

  async createWorkspace(cwd: string, label: string): Promise<string | null> {
    const r = await this.call(["workspace", "create", "--cwd", cwd, "--label", label, "--no-focus"]);
    return r?.workspace?.workspace_id ?? null;
  }

  async closeWorkspace(id: string): Promise<void> {
    await this.call(["workspace", "close", id]);
  }

  async reportTokens(workspaceId: string, tokens: Record<string, string | null>, ttlMs: number): Promise<void> {
    const args = ["workspace", "report-metadata", workspaceId, "--source", "herdr-hub", "--ttl-ms", String(ttlMs)];
    for (const [k, v] of Object.entries(tokens)) {
      if (v === null) args.push("--clear-token", k); else args.push("--token", `${k}=${v}`);
    }
    await this.call(args);
  }

  async createWorktree(o: { repo: string; branch: string; base: string; label: string }): Promise<{ workspaceId: string; paneId: string }> {
    const r = await this.call(["worktree", "create", "--cwd", o.repo, "--branch", o.branch, "--base", o.base, "--label", o.label, "--focus"]);
    return { workspaceId: r.workspace.workspace_id, paneId: r.root_pane.pane_id };
  }

  async createTab(workspaceId: string, label: string): Promise<string> {
    const r = await this.call(["tab", "create", "--workspace", workspaceId, "--label", label, "--focus"]);
    return r.root_pane.pane_id;
  }

  // A brand-new worktree can show Claude's first-run screen; herdr then reports agent_not_ready, which clears itself.
  async startAgent(name: string, paneId: string, agentArgs: string[]): Promise<{ ok: boolean; message: string }> {
    const r = await run(this.bin, [...this.prefix, "agent", "start", name, "--kind", "claude", "--pane", paneId, "--timeout", "120000", "--", ...agentArgs], { timeoutMs: 150_000 });
    const out = `${r.stdout}${r.stderr}`.trim();
    return { ok: r.code === 0 || out.includes("agent_not_ready"), message: out };
  }

  async openPane(entrypoint: string): Promise<void> {
    await this.call(["plugin", "pane", "open", "--plugin", "herdr-hub", "--entrypoint", entrypoint]);
  }

  async notify(title: string, body: string): Promise<void> {
    try { await this.call(["notification", "show", title, "--body", body]); } catch { /* best effort */ }
  }

  async version(): Promise<string | null> {
    const r = await run(this.bin, [...this.prefix, "--version"]);
    return /(\d+\.\d+\.\d+)/.exec(r.stdout)?.[1] ?? null;
  }
```

```ts
export function missingPinned(pinned: Pinned[], openLabels: string[]): Pinned[] {
  const open = new Set(openLabels);
  return pinned.filter((p) => !open.has(p.label));
}

// Serialised with a lock file: a burst of workspace.closed events must not create duplicates.
export function restorePinned(herdr: Herdr, cfg: Config, paths: StatePaths): Promise<string[]> {
  return withFileLock(paths.pinnedLock, async () => {
    if (!cfg.pinned.length) return [];
    const snap = await herdr.snapshot();
    const created: string[] = [];
    for (const p of missingPinned(cfg.pinned, snap.workspaces.map((w) => w.label))) {
      if (!existsSync(p.path)) { log(paths, `pinned "${p.label}" skipped: ${p.path} not found`); continue; }
      await herdr.createWorkspace(p.path, p.label);
      created.push(p.label);
    }
    if (created.length) log(paths, `restored pinned spaces: ${created.join(", ")}`);
    return created;
  });
}
```

```ts
async function locationFor(cwd: string): Promise<Location | null> {
  const hit = cache.get(cwd);
  if (hit && Date.now() - hit.at < TTL) return hit.loc;

  const locations: (Location | null)[] = [];
  for (const cwd of cwds) locations.push(await locationFor(cwd));
  return classify(ws, locations);
```

```ts
// herdr startup hooks must exit, so the updater runs as a detached process guarded by a pid lock.
export function ensureDaemon(paths: StatePaths): boolean {
  const pid = readPidLock(paths.daemonLock);
  if (pid && pidAlive(pid)) return false;
  const script = fileURLToPath(new URL("./daemon.ts", import.meta.url));
  const child = spawn(process.execPath, [script], { detached: true, stdio: "ignore", windowsHide: true, env: process.env });
  child.unref();
  return true;
}
```

```ts
async function tick(): Promise<void> {
  const { config } = loadConfig(configDir());
  const pattern = new RegExp(config.releaseWorkflowPattern, "i");
  const snap = await herdr.snapshot();
  const now = Date.now();
  const cache = readCache(paths);

  const kinds = new Map<string, SpaceKind>();
  for (const ws of snap.workspaces) kinds.set(ws.workspace_id, await spaceKind(ws, snap));

  const slugs = new Set<string>();
  for (const k of kinds.values()) {
    if (k.kind === "other") continue;
    const r = await repoFor(k.repoRoot);
    if (r) slugs.add(`${r.owner}/${r.name}`);
  }
  const due = [...slugs].filter((s) => forceGithub || now - (cache.repos[s]?.fetchedAt ?? 0) >= config.refresh.githubSeconds * 1000);
  if (due.length) {
    const gh = await ghStatus();
    ghProblem = gh === "ok" ? null : gh === "missing" ? "gh not installed" : "gh not signed in";
    if (!ghProblem) {
      for (const slug of due) {
        const [owner, name] = slug.split("/");
        try {
          cache.repos[slug] = { data: await fetchRepoData(owner, name, pattern), fetchedAt: now, error: null };
        } catch (e) {
          const error = (e as Error).message;
          cache.repos[slug] = { data: cache.repos[slug]?.data ?? null, fetchedAt: now, error };
          log(paths, `github ${slug}: ${error}`);
        }
      }
      writeCache(paths, cache);
    }
    forceGithub = false;
  }

  const ttl = Math.max(30_000, config.refresh.localSeconds * 2000);
  for (const ws of snap.workspaces) {
    const k = kinds.get(ws.workspace_id)!;
    if (k.kind === "other") {
      if (Object.keys(ws.tokens ?? {}).some((t) => t.startsWith("hub"))) await herdr.reportTokens(ws.workspace_id, toTokens([]), ttl);
      continue;
    }
    const r = await repoFor(k.repoRoot);
    const entry = r ? cache.repos[`${r.owner}/${r.name}`] : undefined;
    const git = await status(k.folder);
    const agents = snap.agents
      .filter((a) => a.workspace_id === ws.workspace_id)
      .map((a) => ({ kind: a.agent, title: a.terminal_title_stripped || a.name || a.agent, state: a.agent_status }));
    const pr = k.kind === "worktree" && git?.branch ? entry?.data?.prs.find((p) => p.headRefName === git.branch) ?? null : null;
    const lines = buildLines({
      kind: k.kind, git, agents, pr,
      release: k.kind === "main" ? entry?.data?.release ?? null : null,
      ghProblem: r ? ghProblem : null,
      stale: !!entry?.error, now, indent: config.indent,
    });
    await herdr.reportTokens(ws.workspace_id, toTokens(lines), ttl);
  }
}

function waitForTrigger(ms: number): Promise<void> {
  return new Promise((resolve) => {
    let watcher: ReturnType<typeof watch> | undefined;
    const finish = () => { clearTimeout(timer); watcher?.close(); resolve(); };
    const timer = setTimeout(finish, ms);
    try {
      watcher = watch(dirname(paths.trigger), (_event, file) => {
        if (file && String(file) === basename(paths.trigger)) { forceGithub = true; finish(); }
      });
    } catch { /* no fs.watch here: the timer still refreshes */ }
  });
}

async function main(): Promise<void> {
  log(paths, `daemon started (pid ${process.pid})`);
  for (;;) {
    try {
      await tick();
      failures = 0;
    } catch (e) {
      failures++;
      log(paths, `refresh failed (${failures}/3): ${(e as Error).message}`);
      if (failures >= 3) { log(paths, "herdr unreachable, daemon exiting"); process.exit(0); }
    }
    await waitForTrigger(loadConfig(configDir()).config.refresh.localSeconds * 1000);
  }
}

main();
```

```ts
export type Key = { name: "up" | "down" | "enter" | "escape" | "space" | "backspace" | "tab" | "ctrl-c" | "char"; ch?: string };
export interface Item { id: string; label: string; section?: string; hint?: string; note?: string; disabled?: boolean }
export interface ListState { items: Item[]; cursor: number; filter: string; checked: Set<string> }
export type ListResult = { kind: "select"; id: string } | { kind: "done"; ids: string[] } | { kind: "cancel" };

export function parseKeys(s: string): Key[] {
  const out: Key[] = [];
  let i = 0;
  while (i < s.length) {
    const rest = s.slice(i);
    const arrow = /^\x1b(?:\[|O)([AB])/.exec(rest);
    if (arrow) { out.push({ name: arrow[1] === "A" ? "up" : "down" }); i += arrow[0].length; continue; }
    const other = /^\x1b(?:\[[0-9;?]*[A-Za-z~]|O[A-Za-z])/.exec(rest);
    if (other) { i += other[0].length; continue; }
    const c = s[i++];
    if (c === "\x1b") out.push({ name: "escape" });
    else if (c === "\r") { out.push({ name: "enter" }); if (s[i] === "\n") i++; }
    else if (c === "\n") out.push({ name: "enter" });
    else if (c === " ") out.push({ name: "space" });
    else if (c === "\x7f" || c === "\b") out.push({ name: "backspace" });
    else if (c === "\t") out.push({ name: "tab" });
    else if (c === "\x03") out.push({ name: "ctrl-c" });
    else if (c >= " ") out.push({ name: "char", ch: c });
  }
  return out;
}

export function initialList(items: Item[]): ListState {
  return { items, cursor: 0, filter: "", checked: new Set() };
}

export function visibleItems(st: ListState): Item[] {
  const f = st.filter.toLowerCase();
  return f ? st.items.filter((i) => i.label.toLowerCase().includes(f)) : st.items;
}

export function reduceList(st: ListState, key: Key, mode: "single" | "multi"): { st: ListState; result: ListResult | null } {
  const vis = visibleItems(st);
  const cur = vis[st.cursor];
  const same = { st, result: null };
  switch (key.name) {
    case "up": return { st: { ...st, cursor: Math.max(0, st.cursor - 1) }, result: null };
    case "down": return { st: { ...st, cursor: Math.min(Math.max(0, vis.length - 1), st.cursor + 1) }, result: null };
    case "escape": case "ctrl-c": return { st, result: { kind: "cancel" } };
    case "enter":
      if (mode === "multi") return { st, result: { kind: "done", ids: st.items.filter((i) => st.checked.has(i.id)).map((i) => i.id) } };
      return cur && !cur.disabled ? { st, result: { kind: "select", id: cur.id } } : same;
    case "space": {
      if (mode === "single") return { st: { ...st, filter: st.filter + " ", cursor: 0 }, result: null };
      if (!cur || cur.disabled) return same;
      const checked = new Set(st.checked);
      if (checked.has(cur.id)) checked.delete(cur.id); else checked.add(cur.id);
      return { st: { ...st, checked }, result: null };
    }
    case "backspace": return mode === "single" ? { st: { ...st, filter: st.filter.slice(0, -1), cursor: 0 }, result: null } : same;
    case "char": return mode === "single" ? { st: { ...st, filter: st.filter + key.ch, cursor: 0 }, result: null } : same;
    default: return same;
  }
}

const CSI = "\x1b[";
const bold = (s: string) => `${CSI}1m${s}${CSI}0m`;
const dim = (s: string) => `${CSI}2m${s}${CSI}0m`;
const CLEAR = `${CSI}2J${CSI}H`;

export function fit(s: string, w: number): string {
  const chars = [...s];
  return chars.length > w ? chars.slice(0, Math.max(0, w - 1)).join("") + "…" : s;
}

export function renderList(st: ListState, o: { title: string; intro?: string[]; footer: string; width: number; height: number; mode: "single" | "multi" }): string {
  const head = [bold(fit(o.title, o.width)), ...(o.intro ?? []).map((l) => fit(l, o.width))];
  if (st.filter) head.push(dim(fit(`filter: ${st.filter}`, o.width)));
  const body: string[] = [];
  let cursorLine = 0;
  let section: string | undefined;
  const vis = visibleItems(st);
  vis.forEach((it, idx) => {
    if (it.section && it.section !== section) { body.push(dim(fit(it.section.toUpperCase(), o.width))); section = it.section; }
    if (idx === st.cursor) cursorLine = body.length;
    const mark = o.mode === "multi" ? (it.disabled ? "[-] " : st.checked.has(it.id) ? "[x] " : "[ ] ") : "";
    const text = fit(`${idx === st.cursor ? "▸ " : "  "}${mark}${it.label}${it.hint ? "  " + it.hint : ""}`, o.width);
    body.push(it.disabled ? dim(text) : idx === st.cursor ? bold(text) : text);
    if (it.note) body.push(dim(fit(`      ${it.note}`, o.width)));
  });
  if (!vis.length) body.push(dim("  nothing matches"));
  const room = Math.max(1, o.height - head.length - 1);
  const start = Math.max(0, Math.min(cursorLine - room + 2, body.length - room));
  return CLEAR + [...head, ...body.slice(start, start + room)].join("\r\n") + "\r\n" + dim(fit(o.footer, o.width));
}

export function renderMessage(title: string, lines: string[], footer: string, size: { width: number; height: number }): string {
  const room = Math.max(1, size.height - 2);
  return CLEAR + [bold(fit(title, size.width)), ...lines.slice(0, room).map((l) => fit(l, size.width))].join("\r\n") + "\r\n" + dim(fit(footer, size.width));
}
```

```ts
import { initialList, parseKeys, reduceList, renderList, renderMessage } from "./tty.ts";
import type { Item, Key, ListResult, ListState } from "./tty.ts";

export function screen(): { width: number; height: number } {
  return { width: process.stdout.columns || 80, height: process.stdout.rows || 24 };
}

export function readContext(): { workspace_id?: string; workspace_cwd?: string } {
  try { return JSON.parse(process.env.HERDR_PLUGIN_CONTEXT_JSON ?? "{}"); } catch { return {}; }
}

async function withKeys<T>(fn: (next: () => Promise<Key>) => Promise<T>): Promise<T> {
  const stdin = process.stdin;
  const queue: Key[] = [];
  let wake: (() => void) | null = null;
  const onData = (d: Buffer | string) => { queue.push(...parseKeys(d.toString())); wake?.(); };
  if (stdin.isTTY) stdin.setRawMode(true);
  stdin.setEncoding("utf8");
  stdin.on("data", onData);
  stdin.resume();
  process.stdout.write("\x1b[?25l");
  const next = async (): Promise<Key> => {
    while (!queue.length) await new Promise<void>((r) => { wake = r; });
    wake = null;
    return queue.shift()!;
  };
  try { return await fn(next); } finally {
    stdin.off("data", onData);
    if (stdin.isTTY) stdin.setRawMode(false);
    stdin.pause();
    process.stdout.write("\x1b[?25h");
  }
}

export function pickFromList(items: Item[], o: { title: string; intro?: string[]; footer: string; mode: "single" | "multi"; onChar?: (st: ListState, ch: string) => ListState }): Promise<ListResult> {
  return withKeys(async (next) => {
    let st = initialList(items);
    for (;;) {
      process.stdout.write(renderList(st, { ...o, ...screen() }));
      const key = await next();
      if (key.name === "char" && o.mode === "multi" && o.onChar) { st = o.onChar(st, key.ch!); continue; }
      const r = reduceList(st, key, o.mode);
      st = r.st;
      if (r.result) return r.result;
    }
  });
}

export function promptText(title: string, help: string): Promise<string | null> {
  return withKeys(async (next) => {
    let text = "";
    for (;;) {
      process.stdout.write(renderMessage(title, [help, "", `> ${text}█`], "enter to confirm · esc to cancel", screen()));
      const k = await next();
      if (k.name === "escape" || k.name === "ctrl-c") return null;
      if (k.name === "enter") return text;
      if (k.name === "backspace") text = text.slice(0, -1);
      else if (k.name === "space") text += " ";
      else if (k.name === "char") text += k.ch;
    }
  });
}

export function printScreen(title: string, lines: string[]): void {
  process.stdout.write(renderMessage(title, lines, "", screen()));
}

export async function showMessage(title: string, lines: string[]): Promise<void> {
  await withKeys(async (next) => {
    process.stdout.write(renderMessage(title, lines, "press any key", screen()));
    await next();
  });
}
```

```ts
export async function runNewWorktree(workspaceId?: string): Promise<void> {
  const herdr = new Herdr();
  const snap = await herdr.snapshot();
  const ws = snap.workspaces.find((w) => w.workspace_id === workspaceId) ?? snap.workspaces.find((w) => w.focused);
  if (!ws) { await showMessage("New worktree", ["Couldn't tell which space is focused."]); return; }
  const kind = await spaceKind(ws, snap);

    const paneId = await herdr.createTab(ws.workspace_id, "claude");
    await herdr.startAgent(`claude-${Date.now().toString(36)}`, paneId, []);
    return;
  }

  const main = snap.workspaces.find((w) => {
    const k = classify(w, []);
    return k.kind === "main" && k.repoKey === kind.repoKey;
  });
  const spaceLabel = kind.kind === "main" ? ws.label : main?.label ?? basename(kind.repoRoot);
  const { config } = loadConfig(configDir());

  for (;;) {
    const typed = await promptText(`New Claude worktree in ${spaceLabel}`, "Type a short name (e.g. 'stairs fix'), or 'fix/stairs' to pick the branch.");
    if (typed === null) return;
    const plan = planWorktree(typed, config.newWorktree.branchPrefix, spaceLabel);
    if (!plan) continue;
    if (snap.agents.some((a) => a.name === plan.agentName)) {
      await showMessage("Name taken", [`An agent called ${plan.agentName} already exists. Pick another name.`]);
      continue;
    }
    const job: Job = { repo: kind.repoRoot, ...plan };
    const child = spawn(process.execPath, [fileURLToPath(import.meta.url), "--worker", JSON.stringify(job)],

    child.unref();
    printScreen("New worktree", [`Creating ${plan.branch}…`, "It appears in the sidebar when ready."]);
    await new Promise((r) => setTimeout(r, 1200));
    return;
  }
```

```ts
import { basename } from "node:path";
import type { Herdr } from "./io/herdr.ts";
import { baseRef, createdTime, githubRepo, headTime, locate, mergedBranches, status, tipOf, worktrees } from "./io/git.ts";
import { mergedPrBranches, openPrBranches } from "./io/github.ts";
import type { Config } from "./config.ts";
import { findCandidates } from "./model/cleanup.ts";
import type { Candidate } from "./model/cleanup.ts";
import type { SpaceKind } from "./model/classify.ts";
import { normPath } from "./model/paths.ts";
import { spaceKind } from "./locate.ts";

export interface Row extends Candidate { repoRoot: string; repoName: string }

export async function gatherRows(herdr: Herdr, config: Config): Promise<Row[]> {
  const snap = await herdr.snapshot();
  const kinds = new Map<string, SpaceKind>();
  for (const ws of snap.workspaces) kinds.set(ws.workspace_id, await spaceKind(ws, snap));

  const repos = new Map<string, string>(); // repoKey -> repoRoot
  for (const k of kinds.values()) if (k.kind !== "other") repos.set(k.repoKey, k.repoRoot);
  for (const p of config.pinned) {
    const loc = await locate(p.path);
    if (loc) repos.set(normPath(loc.repoRoot), loc.repoRoot);
  }

  const spaces = new Map<string, { workspaceId: string; agentCount: number }>();
  for (const ws of snap.workspaces) {
    const k = kinds.get(ws.workspace_id)!;
    if (k.kind === "worktree") {
      spaces.set(normPath(k.folder), { workspaceId: ws.workspace_id, agentCount: snap.agents.filter((a) => a.workspace_id === ws.workspace_id).length });
    }
  }

  const rows: Row[] = [];
  for (const repoRoot of repos.values()) {
    const list = await worktrees(repoRoot);
    if (list.length < 2) continue;
    const base = await baseRef(repoRoot);
    const gh = await githubRepo(repoRoot);
    let mergedPrs = new Map<string, number>();
    let openPrs = new Set<string>();
    if (gh) {
      mergedPrs = await mergedPrBranches(gh.owner, gh.name).catch(() => new Map<string, number>());
      openPrs = await openPrBranches(gh.owner, gh.name).catch(() => new Set<string>());
    }
    const activityMs = new Map<string, number>();
    const dirty = new Map<string, boolean>();
    for (const w of list) {
      if (w.isMain || w.prunable) continue;
      const key = normPath(w.path);
      const t = Math.max((await headTime(w.path)) ?? 0, createdTime(w.path) ?? 0);
      if (t) activityMs.set(key, t);
      // if git can't read it, treat it as dirty so it can't be removed
      dirty.set(key, (await status(w.path))?.dirty ?? true);
    }
    const candidates = findCandidates({
      worktrees: list, baseTip: await tipOf(repoRoot, base), merged: await mergedBranches(repoRoot, base),
      mergedPrs, openPrBranches: openPrs, activityMs, dirty, spaces, staleDays: config.cleanup.staleDays, now: Date.now(),
    });
    rows.push(...candidates.map((c) => ({ ...c, repoRoot, repoName: basename(repoRoot) })));
  }
  return rows;
}
```

```ts
import { basename } from "node:path";
import { deleteBranch, pruneWorktrees, removeWorktree } from "./io/git.ts";
import type { Row } from "./cleanup-gather.ts";

// The space is closed first: on Windows a shell or agent sitting in the folder stops git removing it.
export async function runCleanupActions(herdr: { closeWorkspace(id: string): Promise<void> }, rows: Row[], deleteBranches: Set<string>): Promise<string[]> {
  const out: string[] = [];
  for (const r of rows) {
    const name = `${r.repoName} · ${basename(r.path)}`;
    try {
      if (r.workspaceId) await herdr.closeWorkspace(r.workspaceId);
      if (r.reason === "missing") {
        const p = await pruneWorktrees(r.repoRoot);
        out.push(p.code === 0 ? `✓ ${name}: pruned` : `✗ ${name}: ${p.stderr.trim()}`);
        continue;
      }
      const rm = await removeWorktree(r.repoRoot, r.path);
      if (rm.code !== 0) { out.push(`✗ ${name}: ${(rm.stderr || rm.stdout).trim()}`); continue; }
      out.push(`✓ ${name}: worktree removed`);
      if (r.branch && deleteBranches.has(r.path)) {
        const b = await deleteBranch(r.repoRoot, r.branch);
        out.push(b.code === 0 ? `✓ ${name}: branch ${r.branch} deleted` : `✗ ${name}: kept branch ${r.branch} (${b.stderr.trim()})`);
      }
    } catch (e) {
      out.push(`✗ ${name}: ${(e as Error).message}`);
    }
  }
  return out;
}
```

```ts
import { basename } from "node:path";
import { Herdr } from "../io/herdr.ts";
import { configDir, loadConfig } from "../config.ts";

import { gatherRows } from "../cleanup-gather.ts";

import type { Item, ListState } from "./tty.ts";
import { pickFromList, printScreen, showMessage } from "./term.ts";

const TITLE = "Clean up workstreams";

function describe(r: Row, deleteBranch: boolean): string {
  const steps = [r.workspaceId ? "close space" : null, r.reason === "missing" ? "prune" : "remove worktree", deleteBranch ? `delete branch ${r.branch}` : null].filter(Boolean);
  return `• ${r.repoName} · ${basename(r.path)}: ${steps.join(", ")}${r.agentCount ? " (stops its agent)" : ""}`;
}

export async function runCleanupUi(): Promise<void> {
  const herdr = new Herdr();
  const { config } = loadConfig(configDir());
  printScreen(TITLE, ["Looking for finished workstreams…"]);
  let rows: Row[];
  try { rows = await gatherRows(herdr, config); } catch (e) {
    await showMessage(TITLE, [`Couldn't list worktrees: ${(e as Error).message}`]);
    return;
  }
  if (!rows.length) { await showMessage(TITLE, ["Nothing to clean up: no merged, stale or missing worktrees."]); return; }

  const deleteBranches = new Set<string>();
  const items: Item[] = rows.map((r) => ({
    id: r.path,
    label: `${r.repoName} · ${basename(r.path)}${r.branch ? ` (${r.branch})` : ""}`,
    note: [r.detail, ...warnings(r)].join(" · "),
    disabled: !r.selectable,
  }));
  const toggleBranch = (st: ListState, ch: string): ListState => {
    if (ch !== "b") return st;
    const it = visibleItems(st)[st.cursor];
    const row = rows.find((r) => r.path === it?.id);
    if (!it || it.disabled || !row?.branch) return st;
    if (deleteBranches.has(it.id)) deleteBranches.delete(it.id); else deleteBranches.add(it.id);
    const checked = new Set(st.checked).add(it.id);
    return { ...st, checked, items: st.items.map((x) => x.id === it.id ? { ...x, hint: deleteBranches.has(it.id) ? "+ delete branch" : undefined } : x) };
  };

  const picked = await pickFromList(items, { title: `${TITLE} · ${rows.length} found`, footer: "↑↓ move · space tick · b also delete branch · enter continue · esc cancel", mode: "multi", onChar: toggleBranch });
  if (picked.kind !== "done" || !picked.ids.length) return;
  const chosen = rows.filter((r) => picked.ids.includes(r.path));

  const confirm = await pickFromList(
    [{ id: "go", label: `Clean up ${chosen.length} workstream${chosen.length === 1 ? "" : "s"}` }, { id: "cancel", label: "Cancel" }],
    { title: "Confirm", intro: chosen.map((r) => describe(r, deleteBranches.has(r.path))).concat([""]), footer: "enter choose · esc cancel", mode: "single" });
  if (confirm.kind !== "select" || confirm.id !== "go") return;

  printScreen(TITLE, ["Working…"]);
  const results = await runCleanupActions(herdr, chosen, deleteBranches);
  touchTrigger(statePaths());
  await showMessage("Done", results);
}
```

```ts
export async function runChecks(): Promise<Check[]> {
  const checks: Check[] = [];
  const node = process.versions.node;
  checks.push({ name: "Node", ok: compareVersions(node, "24.0.0") >= 0, detail: `${node} (needs 24 or newer)` });

  const git = await run("git", ["--version"]);
  checks.push({ name: "git", ok: git.code === 0, detail: git.code === 0 ? git.stdout.trim() : "not found" });

  const gh = await ghStatus();
  checks.push({ name: "GitHub CLI", ok: gh === "ok", detail: gh === "ok" ? "signed in" : gh === "missing" ? "not installed (https://cli.github.com)" : "run: gh auth login" });

  const hv = await new Herdr().version();
  checks.push({ name: "herdr", ok: !!hv && compareVersions(hv, "0.9.0") >= 0, detail: hv ? `${hv} (needs 0.9.0 or newer)` : "not found" });

  const toml = herdrConfigPath(configDir());
  checks.push(...snippetChecks(existsSync(toml) ? readFileSync(toml, "utf8") : null).map((c) => c.ok ? { ...c, detail: "ok" } : c));

  const { errors } = loadConfig(configDir());
  checks.push({ name: "config.json", ok: !errors.length, detail: errors.length ? errors.join("; ") : "ok" });

  const pid = readPidLock(statePaths().daemonLock);
  checks.push({ name: "sidebar updater", ok: !!pid && pidAlive(pid), detail: pid && pidAlive(pid) ? `running (pid ${pid})` : "not running: use Refresh now" });
  return checks;
}

export async function runDoctor(): Promise<void> {
  const checks = await runChecks();
  await showMessage("Check setup", checks.map((c) => `${c.ok ? "✓" : "✗"} ${c.name}: ${c.detail}`));
}
```
